    function buildInviteUrl(code) {
      const clean = String(code || '').trim().toUpperCase();
      if (!clean) return '';
      try {
        const url = new URL(window.location.href);
        url.searchParams.set('room', clean);
        return url.toString();

        return `${base}?room=${encodeURIComponent(clean)}`;

        if (code) url.searchParams.set('room', String(code).trim().toUpperCase());
        else url.searchParams.delete('room');

    function copyTextToClipboard(text, successMsg) {
      const val = String(text || '');
      if (!val) return;
      if (navigator.clipboard && typeof navigator.clipboard.writeText === 'function') {
        navigator.clipboard.writeText(val).then(
          () => setLiveBanner(successMsg, false),
          () => legacyCopy(val, successMsg)
        );
      } else {
        legacyCopy(val, successMsg);
      }
    }

    function legacyCopy(val, successMsg) {
      try {
        const ta = document.createElement('textarea');
        ta.value = val;
        ta.setAttribute('readonly', '');
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        document.execCommand('copy');
        document.body.removeChild(ta);
        setLiveBanner(successMsg, false);
      } catch (_) {
        setLiveBanner(`انسخ الرابط يدوياً: ${val}`, false);
      }

    function createShuffledDeck() {
      const deck = [];
      let n = 1;
      for (const suit of SUITS) {
        for (const rank of RANKS) {
          deck.push({
            id: `c_${n++}_${suit.key}_${rank.code}_${Math.random().toString(36).slice(2, 5)}`,
            suitKey: suit.key,
            suitSymbol: suit.symbol,
            suitName: suit.name,
            isRed: suit.isRed,
            rankCode: rank.code,
            rankLabel: rank.label,
            points: rank.points
          });
        }
      }
      for (let i = deck.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [deck[i], deck[j]] = [deck[j], deck[i]];
      }
      return deck;
    }

    function calculateHandSum(hand) {
      if (!Array.isArray(hand)) return 0;
      return hand.reduce((sum, card) => sum + (Number(card.points) || 0), 0);
    }

    function describeCards(cards) {
      return cards.map((c) => `${c.rankLabel}${c.suitSymbol}`).join(' ، ');
    }

    // Allows single card or multiple cards sharing the same rank OR same suit
    function areCardsValidDiscardGroup(cards) {
      if (!Array.isArray(cards) || cards.length === 0) return false;
      if (cards.length === 1) return true;
      const sameRank = cards.every((c) => c.rankCode === cards[0].rankCode);
      const sameSuit = cards.every((c) => c.suitKey === cards[0].suitKey);
      return sameRank || sameSuit;
    }

    function getPreviousPlayerIndex(idx, total = roomState.players.length) {
      return (idx - 1 + total) % total;

    function getNextPlayerIndex(idx, total = roomState.players.length) {

    }

    function getMyPlayerIndex() {

    function getMyPlayer() {
      const idx = getMyPlayerIndex();
      return idx >= 0 ? roomState.players[idx] : null;

    function canDeclareInziliForMe() {
      const myIdx = getMyPlayerIndex();
      const me = myIdx >= 0 ? roomState.players[myIdx] : null;
      if (!me || !roomState.matchStarted || roomState.roundEnded || roomState.matchFinished) return false;
      if (roomState.currentPlayerIndex !== myIdx) return false;
      if (roomState.turnPhase !== 'DISCARD' || roomState.hasDiscardedThisTurn) return false;
      return calculateHandSum(me.hand) <= 12;
    }

    /* Real-Time Room Sync Layer (Supabase Realtime + BroadcastChannel Fallback) */
    function serializeState() {
      return {
        roomCode: roomState.roomCode,
        hostId: roomState.hostId,
        maxPlayers: roomState.maxPlayers,
        lossLimit: roomState.lossLimit,
        matchStarted: roomState.matchStarted,
        matchFinished: roomState.matchFinished,
        loserId: roomState.loserId,

        bannerText: roomState.bannerText,
        bannerIsJoinFlash: roomState.bannerIsJoinFlash
      };
    }

    function applyState(snap) {
      if (!snap || snap.roomCode !== roomState.roomCode) return;
      roomState.hostId = snap.hostId;
      roomState.maxPlayers = snap.maxPlayers || roomState.maxPlayers;
      roomState.lossLimit = snap.lossLimit || roomState.lossLimit;
      roomState.matchStarted = Boolean(snap.matchStarted);
      roomState.matchFinished = Boolean(snap.matchFinished);
      roomState.loserId = snap.loserId ?? null;
      roomState.roundNumber = snap.roundNumber || 1;
      roomState.roundStarterIndex = snap.roundStarterIndex || 0;
      roomState.currentPlayerIndex = snap.currentPlayerIndex || 0;
      roomState.turnPhase = snap.turnPhase || 'DISCARD';
      roomState.hasDiscardedThisTurn = Boolean(snap.hasDiscardedThisTurn);
      roomState.roundEnded = Boolean(snap.roundEnded);
      roomState.players = Array.isArray(snap.players) ? snap.players : [];
      roomState.drawPile = Array.isArray(snap.drawPile) ? snap.drawPile : [];
      if (snap.bannerText) {
        setLiveBanner(snap.bannerText, Boolean(snap.bannerIsJoinFlash));
      }

      const me = getMyPlayer();
      const myIds = new Set((me ? me.hand : []).map((c) => c.id));
      roomState.selectedCardIds = roomState.selectedCardIds.filter((id) => myIds.has(id));

      renderAll();
    }

    function sendEvent(type, payload = {}) {
      if (!transport.channel || !roomState.roomCode) return;
      const packet = {
        type,
        senderId: roomState.localPlayerId,
        roomCode: roomState.roomCode,
        payload
      };
      if (transport.mode === 'BROADCAST') {
        transport.channel.postMessage(packet);
      } else if (transport.mode === 'SUPABASE') {
        transport.channel.send({
          type: 'broadcast',
          event: 'room_sync',
          payload: packet
        });
      }
    }

    function broadcastRoomStateIfHost() {
      if (!isHost()) return;
      sendEvent('SYNC_STATE', { snapshot: serializeState() });
    }

    async function connectChannel(code) {
      if (transport.channel) {
        try {
          if (transport.mode === 'BROADCAST') transport.channel.close();
          else if (transport.mode === 'SUPABASE' && transport.supabase) {
            await transport.supabase.removeChannel(transport.channel);
          }
        } catch (_) {}
        transport.channel = null;
      }

      const url = (document.getElementById('supabase-url-input')?.value || '').trim();
      const key = (document.getElementById('supabase-key-input')?.value || '').trim();
      const badge = document.getElementById('connection-badge');

      if (url && key && window.supabase && typeof window.supabase.createClient === 'function') {
        transport.mode = 'SUPABASE';
        transport.supabase = window.supabase.createClient(url, key);
        const ch = transport.supabase.channel(`inzili_room_${code}`, {
          config: { broadcast: { self: false } }
        });
        ch.on('broadcast', { event: 'room_sync' }, ({ payload }) => {
          safeRun(() => handleIncomingEvent(payload));
        });
        await ch.subscribe((status) => {
          if (badge) {
            badge.textContent = status === 'SUBSCRIBED' ? 'الاتصال: Supabase متصل' : `الاتصال: ${status}`;
          }
        });
        transport.channel = ch;
      } else {
        transport.mode = 'BROADCAST';
        const ch = new BroadcastChannel(`inzili_room_${code}`);
        ch.onmessage = (event) => {
          safeRun(() => handleIncomingEvent(event.data));
        };
        transport.channel = ch;
        if (badge) {
          badge.textContent = 'الاتصال: محاكاة تبويبات';
        }
      }
    }

    function handleIncomingEvent(msg) {
      if (!msg || msg.senderId === roomState.localPlayerId || msg.roomCode !== roomState.roomCode) return;

      if (msg.type === 'SYNC_STATE' && msg.payload && msg.payload.snapshot) {
        applyState(msg.payload.snapshot);
        return;
      }

      if (isHost()) {
        if (msg.type === 'PLAYER_JOIN') {
          const { playerId, name } = msg.payload || {};
          if (!playerId) return;
          const existing = roomState.players.find((p) => p.id === playerId);
          if (!existing && roomState.players.length < roomState.maxPlayers) {
            const joinedName = (name || '').trim() || `لاعب ${roomState.players.length + 1}`;
            roomState.players.push({
              id: playerId,
              name: joinedName,
              score: 0,
              hand: [],
              leftDiscard: null,
              leftSourceLabel: null
            });
            setLiveBanner(`انضم ${joinedName} إلى الغرفة (${formatArabicNumber(roomState.players.length)}/${formatArabicNumber(roomState.maxPlayers)}).`, true);
          }
          broadcastRoomStateIfHost();
          renderAll();
        } else if (msg.type === 'PLAYER_LEAVE') {
          const idx = roomState.players.findIndex((p) => p.id === msg.payload?.playerId);
          if (idx >= 0) {
            const leftName = roomState.players[idx].name;
            roomState.players.splice(idx, 1);
            if (roomState.players.length > 0) {
              roomState.currentPlayerIndex = roomState.currentPlayerIndex % roomState.players.length;
              roomState.roundStarterIndex = roomState.roundStarterIndex % roomState.players.length;
            }
            setLiveBanner(`غادر ${leftName} الغرفة.`, false);
            broadcastRoomStateIfHost();
            renderAll();
          }
        } else if (msg.type === 'REQ_DISCARD') {
          applyDiscardMove(msg.payload.playerIndex, msg.payload.cardIds);
          renderAll();
        } else if (msg.type === 'REQ_DRAW') {
          applyDrawMove(msg.payload.playerIndex, msg.payload.source);
          renderAll();
        } else if (msg.type === 'REQ_INZILI') {
          applyInziliMove(msg.payload.playerIndex);
          renderAll();
        }
      }
    }

    function dealRoundForRoom(roundNum, starterIdx) {
      const deck = createShuffledDeck();
      const count = roomState.players.length;
      roomState.matchStarted = true;
      roomState.roundNumber = roundNum;
      roomState.roundStarterIndex = starterIdx % count;
      roomState.currentPlayerIndex = roomState.roundStarterIndex;
      roomState.turnPhase = 'DISCARD';
      roomState.hasDiscardedThisTurn = false;
      roomState.roundEnded = false;
      roomState.selectedCardIds = [];

      for (let i = 0; i < count; i++) {
        const p = roomState.players[i];
        p.hand = [];
        p.leftDiscard = null;
        p.leftSourceLabel = null;
        for (let c = 0; c < 4; c++) {
          if (deck.length > 0) p.hand.push(deck.pop());
        }
      }

      roomState.drawPile = deck;

      const starter = roomState.players[roomState.roundStarterIndex];
      if (starter && roomState.drawPile.length > 0) {
        starter.leftDiscard = roomState.drawPile.pop();
        starter.leftSourceLabel = 'من اللعبة';
      }

      for (let step = 1; step < count; step++) {
        const idx = (roomState.roundStarterIndex + step) % count;
        const prevIdx = getPreviousPlayerIndex(idx, count);
        const player = roomState.players[idx];
        const prevPlayer = roomState.players[prevIdx];
        if (!player) continue;
        if (prevPlayer && prevPlayer.leftDiscard) {
          player.leftDiscard = prevPlayer.leftDiscard;
          player.leftSourceLabel = `من ${prevPlayer.name}`;
        } else if (roomState.drawPile.length > 0) {
          player.leftDiscard = roomState.drawPile.pop();
          player.leftSourceLabel = 'من اللعبة';
        }
      }
    }

    function hostStartOrNextRound() {
      // Strictly host-only authority
      if (!isHost()) {
        setLiveBanner('فقط اللاعب الأساسي يمكنه بدء اللعب أو بدء جولة جديدة.', false);
        return;
      }

      if (roomState.players.length < 2) {
        setLiveBanner('يلزم وجود لاعبين اثنين على الأقل في الغرفة لبدء اللعب.', false);
        return;
      }

      if (!roomState.matchStarted || roomState.matchFinished) {
        for (const p of roomState.players) {
          p.score = 0;
          p.hand = [];
          p.leftDiscard = null;
          p.leftSourceLabel = null;
        }
        roomState.matchFinished = false;
        roomState.loserId = null;
        dealRoundForRoom(1, 0);
        setLiveBanner(`بدأ اللاعب الأساسي اللعب — الجولة ١ والدور عند ${roomState.players[roomState.currentPlayerIndex].name}.`, false);
      } else if (roomState.roundEnded) {
        const nextStarter = getNextPlayerIndex(roomState.roundStarterIndex);
        dealRoundForRoom(roomState.roundNumber + 1, nextStarter);
        setLiveBanner(`بدأ اللاعب الأساسي الجولة ${formatArabicNumber(roomState.roundNumber)} — الدور عند ${roomState.players[roomState.currentPlayerIndex].name}.`, false);
      }

      broadcastRoomStateIfHost();
      renderAll();
    }

    /* Live Turn Moves Visible to Everyone */
    function toggleSelectCard(cardId) {
      const myIdx = getMyPlayerIndex();
      if (myIdx < 0 || !roomState.matchStarted || roomState.roundEnded || roomState.matchFinished) return;
      if (roomState.currentPlayerIndex !== myIdx) {
        setLiveBanner(`ليس دورك الآن — الدور عند ${roomState.players[roomState.currentPlayerIndex]?.name || 'لاعب آخر'}.`, false);
        return;
      }
      if (roomState.turnPhase !== 'DISCARD') {
        setLiveBanner('أنت في مرحلة السحب الآن: اسحب من يسارك أو من الرزمة المقلوبة.', false);
        return;
      }

      const pos = roomState.selectedCardIds.indexOf(cardId);
      if (pos >= 0) roomState.selectedCardIds.splice(pos, 1);
      else roomState.selectedCardIds.push(cardId);

      renderMyHand();
      renderActionButtons();
    }

    function clearSelectedCards() {
      roomState.selectedCardIds = [];
      renderMyHand();
      renderActionButtons();
    }

    function applyDiscardMove(playerIndex, cardIds) {
      const player = roomState.players[playerIndex];
      if (!player || !roomState.matchStarted || roomState.roundEnded || roomState.matchFinished) return false;
      if (roomState.currentPlayerIndex !== playerIndex || roomState.turnPhase !== 'DISCARD') return false;

      if (!Array.isArray(cardIds) || cardIds.length === 0) {
        setLiveBanner('اختر بطاقة واحدة أو أكثر متشابهة للرمي.', false);
        return false;
      }

      const chosen = player.hand.filter((c) => cardIds.includes(c.id));
      if (chosen.length !== cardIds.length || !areCardsValidDiscardGroup(chosen)) {
        setLiveBanner('يجب أن تكون البطاقات المرمية متشابهة في الرقم أو الشكل.', false);
        return false;
      }

      const removeSet = new Set(cardIds);
      player.hand = player.hand.filter((c) => !removeSet.has(c.id));

      const topCard = chosen[chosen.length - 1];
      const nextPlayer = roomState.players[getNextPlayerIndex(playerIndex)];
      if (nextPlayer) {
        nextPlayer.leftDiscard = topCard;
        nextPlayer.leftSourceLabel = `من ${player.name}`;
      }

      if (player.id === roomState.localPlayerId) {
        roomState.selectedCardIds = [];
      }
      roomState.hasDiscardedThisTurn = true;
      roomState.turnPhase = 'DRAW';

      setLiveBanner(`رمى ${player.name} (${describeCards(chosen)}) — الآن مرحلة السحب.`, false);
      broadcastRoomStateIfHost();
      return true;
    }

    function triggerDiscard() {
      const myIdx = getMyPlayerIndex();
      const me = myIdx >= 0 ? roomState.players[myIdx] : null;
      if (!me || roomState.currentPlayerIndex !== myIdx) return;

      const ids = [...roomState.selectedCardIds];
      if (!isHost()) {
        const chosen = me.hand.filter((c) => ids.includes(c.id));
        if (!areCardsValidDiscardGroup(chosen)) {
          setLiveBanner('يجب أن تكون البطاقات المرمية متشابهة في الرقم أو الشكل.', false);
          return;
        }
        roomState.selectedCardIds = [];
        sendEvent('REQ_DISCARD', { playerIndex: myIdx, cardIds: ids });
        renderAll();
        return;
      }

      if (applyDiscardMove(myIdx, ids)) {
        renderAll();
      }
    }

    function applyDrawMove(playerIndex, source) {
      const player = roomState.players[playerIndex];
      if (!player || !roomState.matchStarted || roomState.roundEnded || roomState.matchFinished) return false;
      if (roomState.currentPlayerIndex !== playerIndex || roomState.turnPhase !== 'DRAW') return false;

      let drawn = null;
      const fromLeft = source === 'LEFT';

      if (fromLeft) {
        if (!player.leftDiscard) {
          setLiveBanner('لا توجد بطاقة متاحة على يسارك.', false);
          return false;
        }
        drawn = player.leftDiscard;
        player.leftDiscard = null;
        player.leftSourceLabel = null;
      } else {
        if (roomState.drawPile.length === 0) {
          const fresh = createShuffledDeck();
          const used = new Set();
          for (const p of roomState.players) {
            for (const c of p.hand) used.add(c.id);
            if (p.leftDiscard) used.add(p.leftDiscard.id);
          }
          roomState.drawPile = fresh.filter((c) => !used.has(c.id));
          if (roomState.drawPile.length === 0) roomState.drawPile = fresh;
        }
        drawn = roomState.drawPile.pop();
      }

      if (!drawn) return false;
      player.hand.push(drawn);

      roomState.currentPlayerIndex = getNextPlayerIndex(playerIndex);
      roomState.turnPhase = 'DISCARD';
      roomState.hasDiscardedThisTurn = false;
      if (player.id === roomState.localPlayerId) {
        roomState.selectedCardIds = [];
      }

      const nextP = roomState.players[roomState.currentPlayerIndex];
      const drawDesc = fromLeft ? `من يساره (${drawn.rankLabel}${drawn.suitSymbol})` : 'من الرزمة';
      setLiveBanner(`سحب ${player.name} ${drawDesc} — الدور الآن عند ${nextP ? nextP.name : '—'}.`, false);
      broadcastRoomStateIfHost();
      return true;
    }

    function triggerDraw(source) {
      const myIdx = getMyPlayerIndex();
      const me = myIdx >= 0 ? roomState.players[myIdx] : null;
      if (!me || roomState.currentPlayerIndex !== myIdx) return;

      if (!isHost()) {
        sendEvent('REQ_DRAW', { playerIndex: myIdx, source });
        return;
      }

      if (applyDrawMove(myIdx, source)) {
        renderAll();
      }
    }

    function applyInziliMove(callerIndex) {
      if (!roomState.matchStarted || roomState.roundEnded || roomState.matchFinished) return false;
      const caller = roomState.players[callerIndex];
      if (!caller) return false;

      if (roomState.currentPlayerIndex !== callerIndex || roomState.turnPhase !== 'DISCARD' || roomState.hasDiscardedThisTurn) {
        setLiveBanner('إعلان «انزلي!» متاح عند حلول دورك وقبل الرمي فقط.', false);
        return false;
      }

      const callerSum = calculateHandSum(caller.hand);
      if (callerSum > 12) {
        setLiveBanner(`مجموع يد ${caller.name} (${formatArabicNumber(callerSum)}) أكبر من ١٢.`, false);
        return false;
      }

      roomState.roundEnded = true;
      roomState.turnPhase = 'ROUND_END';
      roomState.selectedCardIds = [];

      const otherSums = roomState.players
        .filter((_, idx) => idx !== callerIndex)
        .map((p) => calculateHandSum(p.hand));
      const minOther = otherSums.length > 0 ? Math.min(...otherSums) : Infinity;

      if (callerSum < minOther) {
        roomState.players.forEach((p, idx) => {
          if (idx !== callerIndex) {
            p.score += calculateHandSum(p.hand);
          }
        });
        setLiveBanner(`أعلن ${caller.name} «انزلي!» بمجموع ${formatArabicNumber(callerSum)} وحقق أقل مجموع (+٠).`, false);
      } else {
        const penalty = callerSum * 2 + Math.max(0, callerSum - minOther);
        caller.score += penalty;
        roomState.players.forEach((p, idx) => {
          if (idx !== callerIndex) {
            p.score += calculateHandSum(p.hand);
          }
        });
        setLiveBanner(`أعلن ${caller.name} «انزلي!» بمجموع ${formatArabicNumber(callerSum)} وتم تدبيله (+${formatArabicNumber(penalty)}).`, false);
      }

      const loser = roomState.players.find((p) => p.score >= roomState.lossLimit);
      if (loser) {
        roomState.matchFinished = true;
        roomState.loserId = loser.id;
        setLiveBanner(`انتهت المباراة: وصل ${loser.name} إلى حد الخسارة (${formatArabicNumber(roomState.lossLimit)}).`, false);
      }

      broadcastRoomStateIfHost();
      renderAll();
      return true;
    }

    function triggerInzili() {
      const myIdx = getMyPlayerIndex();
      if (myIdx < 0 || !canDeclareInziliForMe()) {
        setLiveBanner('يُسمح بإعلان «انزلي!» عند حلول دورك وقبل الرمي إذا كان مجموع يدك ١٢ أو أقل.', false);
        return;
      }

      if (!isHost()) {
        sendEvent('REQ_INZILI', { playerIndex: myIdx });
        return;
      }

      applyInziliMove(myIdx);
    }

      if (m) {
        m.classList.remove('hidden');
        m.classList.add('flex');
      }
    }

      if (m) {
        m.classList.add('hidden');
        m.classList.remove('flex');
      }
    }

    /* Rendering (Cards & Hand Total Always Visible) */
    function renderLossLimitPill() {
      const pill = document.getElementById('loss-limit-pill');
      if (pill) pill.textContent = `حد الخسارة: ${formatArabicNumber(roomState.lossLimit)}`;
    }

    function renderHostControls() {
      const hostPanel = document.getElementById('host-controls-panel');
      const guestPanel = document.getElementById('guest-waiting-panel');
      const roleBadge = document.getElementById('role-badge');
      const hostView = isHost();

      if (roleBadge) {
        roleBadge.textContent = hostView ? 'اللاعب الأساسي' : 'لاعب منضم';
      }

      // Invite link and room settings are strictly shown to the host and hidden from guests
      if (hostPanel) hostPanel.classList.toggle('hidden', !hostView);
      if (guestPanel) guestPanel.classList.toggle('hidden', hostView || roomState.matchStarted);

      if (!hostView) return;

      const codeLabel = document.getElementById('host-room-code-label');
      const urlText = document.getElementById('host-invite-url-text');
      if (codeLabel) codeLabel.textContent = roomState.roomCode || '—';
      if (urlText) urlText.textContent = buildInviteUrl(roomState.roomCode);

      const startBtn = document.getElementById('btn-start-or-next-round');
      if (startBtn) {
        const canStartOrNext = !roomState.matchStarted || roomState.roundEnded || roomState.matchFinished;
        startBtn.disabled = !canStartOrNext || roomState.players.length < 2;
        startBtn.textContent = !roomState.matchStarted
          ? 'بدء اللعب'
          : roomState.matchFinished
          ? 'مباراة جديدة'
          : 'جولة جديدة';
      }

      const locked = isHostSettingsLocked();
      const activeCls = 'bg-amber-400/25 border-amber-300 text-amber-200';
      const inactiveCls = 'bg-black/30 border-amber-400/25 text-amber-100/75 hover:border-amber-400/50';

      for (const lim of [250, 500, 750]) {
        const btn = document.getElementById(`host-limit-${lim}`);
        if (btn) {
          btn.className = `px-3 py-1 rounded-xl border text-xs font-black cursor-pointer transition ${
            roomState.lossLimit === lim ? activeCls : inactiveCls
          }`;
          btn.disabled = locked;
        }
      }

      const maxSelect = document.getElementById('room-max-players-select');
      if (maxSelect) {
        maxSelect.value = String(roomState.maxPlayers);
        maxSelect.disabled = locked;
      }
    }

    function renderPlayersStrip() {
      const strip = document.getElementById('players-strip');
      if (!strip) return;
      strip.innerHTML = '';

      roomState.players.forEach((p, idx) => {
        const isTurn = roomState.matchStarted && !roomState.roundEnded && idx === roomState.currentPlayerIndex;
        const isLoser = roomState.loserId === p.id;
        const sum = calculateHandSum(p.hand);

        const card = document.createElement('div');
        card.className = `rounded-2xl p-3.5 border transition ${
          isLoser
            ? 'bg-red-950/85 border-red-400/70'
            : isTurn
            ? 'burgundy-panel border-amber-300 ring-2 ring-amber-300/40'
            : 'luxury-surface border-amber-400/25'
        }`;

        const roleLabel = p.id === roomState.hostId ? 'أساسي' : p.id === roomState.localPlayerId ? 'أنت' : 'لاعب';
        const detailText =
          p.id === roomState.localPlayerId || roomState.roundEnded
            ? `مجموع اليد: ${formatArabicNumber(sum)}`
            : `البطاقات: ${formatArabicNumber(p.hand.length)}`;

        card.innerHTML = `
          <div class="flex items-center justify-between gap-2">
            <span class="font-black text-sm text-amber-200 truncate">${p.name}</span>
            <span class="px-2 py-0.5 rounded-lg bg-black/35 border border-amber-400/30 text-[11px] font-bold text-amber-100">${roleLabel}</span>
          </div>
          <div class="mt-2 flex items-baseline justify-between gap-2">
            <span class="text-xs font-bold text-emerald-100/85">${detailText}</span>
            <span class="text-base font-black text-amber-300">${formatArabicNumber(p.score)} <span class="text-xs">نقطة</span></span>
          </div>
        `;
        strip.appendChild(card);
      });
    }

    function createCardFaceEl(card, selected, clickable, onClick) {
      const el = document.createElement('div');
      el.className = `playing-card ${selected ? 'selected' : ''} ${clickable ? 'cursor-pointer' : ''}`;
      const cls = card.isRed ? 'suit-red' : 'suit-black';

      el.innerHTML = `
        <div class="flex items-center justify-between text-xs sm:text-sm font-black leading-none ${cls}">
          <span>${card.rankLabel}</span><span>${card.suitSymbol}</span>
        </div>
        <div class="text-center text-2xl sm:text-3xl leading-none font-black ${cls}">${card.suitSymbol}</div>
        <div class="flex items-center justify-between text-[11px] font-extrabold leading-none ${cls}">
          <span>${formatArabicNumber(card.points)}ن</span><span>${card.rankLabel}</span>
        </div>
      `;

      if (clickable && typeof onClick === 'function') {
        el.addEventListener('click', (e) => {
          e.preventDefault();
          safeRun(() => onClick(card.id));
        });
      }
      return el;
    }

    function createCardBackEl(label = 'انزلي') {
      const el = document.createElement('div');
      el.className = 'playing-card card-back items-center justify-center text-center';
      el.innerHTML = `
        <div class="w-10 h-10 rounded-full border border-amber-300/60 flex items-center justify-center text-lg font-black">♠</div>
        <span class="text-xs font-black tracking-wide text-amber-200">${label}</span>
      `;
      return el;
    }

    function renderPiles() {
      const myIdx = getMyPlayerIndex();
      const me = myIdx >= 0 ? roomState.players[myIdx] : null;
      const deckSlot = document.getElementById('draw-deck-slot');
      const deckLabel = document.getElementById('deck-count-label');
      const leftSlot = document.getElementById('left-discard-slot');
      const leftLabel = document.getElementById('left-discard-label');
      const leftHint = document.getElementById('left-discard-hint');

      if (deckLabel) {
        deckLabel.textContent = `الرزمة المقلوبة (${formatArabicNumber(roomState.drawPile.length)})`;
      }
      if (deckSlot) {
        deckSlot.innerHTML = '';
        deckSlot.appendChild(createCardBackEl('سحب'));
      }

      if (leftSlot) {
        leftSlot.innerHTML = '';
        const srcLabel = me && me.leftSourceLabel ? me.leftSourceLabel : 'من السابق';
        if (leftLabel) leftLabel.textContent = `من يسارك (${srcLabel})`;

        const leftCard = me ? me.leftDiscard : null;
        if (leftCard) {
          const canDrawLeft =
            roomState.matchStarted &&
            !roomState.roundEnded &&
            !roomState.matchFinished &&
            roomState.currentPlayerIndex === myIdx &&
            roomState.turnPhase === 'DRAW';

          leftSlot.appendChild(createCardFaceEl(leftCard, false, canDrawLeft, () => triggerDraw('LEFT')));
          if (leftHint) {
            leftHint.textContent = `ورقة يسارك (${srcLabel}): ${leftCard.rankLabel}${leftCard.suitSymbol}`;
          }
        } else {
          const empty = document.createElement('div');
          empty.className = 'playing-card items-center justify-center text-center bg-black/25 border-dashed border-amber-400/35 text-amber-200/60 text-xs font-bold';
          empty.textContent = 'فارغ';
          leftSlot.appendChild(empty);
          if (leftHint) leftHint.textContent = 'لا توجد ورقة على يسارك حالياً';
        }
      }
    }

    function renderMyHand() {
      const container = document.getElementById('my-hand-container');
      const titleEl = document.getElementById('my-hand-title');
      const sumEl = document.getElementById('my-hand-sum-label');
      if (!container) return;

      container.innerHTML = '';
      const myIdx = getMyPlayerIndex();
      const me = myIdx >= 0 ? roomState.players[myIdx] : null;
      const hand = me ? me.hand : [];
      const sum = calculateHandSum(hand);

      if (titleEl) titleEl.textContent = `بطاقاتك (${me ? me.name : '—'})`;
      // Always show hand total clearly without any hiding option
      if (sumEl) sumEl.textContent = `مجموع اليد: ${formatArabicNumber(sum)} (الهدف ≤ ١٢)`;

      const canSelect =
        roomState.matchStarted &&
        !roomState.roundEnded &&
        !roomState.matchFinished &&
        roomState.currentPlayerIndex === myIdx &&
        roomState.turnPhase === 'DISCARD';

      hand.forEach((card) => {
        const selected = roomState.selectedCardIds.includes(card.id);
        container.appendChild(createCardFaceEl(card, selected, canSelect, toggleSelectCard));
      });
    }

    function renderActionButtons() {
      const myIdx = getMyPlayerIndex();
      const me = myIdx >= 0 ? roomState.players[myIdx] : null;
      const isMyTurn =
        myIdx >= 0 &&
        roomState.matchStarted &&
        !roomState.roundEnded &&
        !roomState.matchFinished &&
        roomState.currentPlayerIndex === myIdx;

      const selectedCards = me ? me.hand.filter((c) => roomState.selectedCardIds.includes(c.id)) : [];
      const canDiscard = isMyTurn && roomState.turnPhase === 'DISCARD' && areCardsValidDiscardGroup(selectedCards);
      const canInzili = canDeclareInziliForMe();

      const discardBtn = document.getElementById('btn-discard');
      const clearBtn = document.getElementById('btn-clear-selection');
      const inziliBtn = document.getElementById('btn-inzili');

      if (discardBtn) discardBtn.disabled = !canDiscard;
      if (clearBtn) clearBtn.disabled = !isMyTurn || roomState.selectedCardIds.length === 0;
      if (inziliBtn) inziliBtn.disabled = !canInzili;
    }

    function renderRoomBadges() {
      const roundBadge = document.getElementById('round-badge');
      const turnBadge = document.getElementById('turn-badge');
      const phaseBadge = document.getElementById('phase-badge');
      const playersBadge = document.getElementById('players-count-badge');
      const activePlayer = roomState.players[roomState.currentPlayerIndex];

      if (roundBadge) roundBadge.textContent = `الجولة ${formatArabicNumber(roomState.roundNumber)}`;
      if (turnBadge) {
        turnBadge.textContent = roomState.matchFinished
          ? 'انتهت المباراة'
          : roomState.roundEnded
          ? 'انتهت الجولة'
          : !roomState.matchStarted
          ? 'بانتظار البدء'
          : `الدور: ${activePlayer ? activePlayer.name : '—'}`;
      }
      if (phaseBadge) {
        if (!roomState.matchStarted) phaseBadge.textContent = 'المرحلة: انتظار اللاعب الأساسي';
        else if (roomState.matchFinished) phaseBadge.textContent = 'المرحلة: نهاية المباراة';
        else if (roomState.roundEnded) phaseBadge.textContent = 'المرحلة: نهاية الجولة';
        else phaseBadge.textContent = roomState.turnPhase === 'DISCARD' ? 'المرحلة: الرمي' : 'المرحلة: السحب';
      }
      if (playersBadge) {
        playersBadge.textContent = `اللاعبون: ${formatArabicNumber(roomState.players.length)} / ${formatArabicNumber(roomState.maxPlayers)}`;
      }
      renderLossLimitPill();
    }

    function renderAll() {
      renderHostControls();
      renderPlayersStrip();
      renderPiles();
      renderMyHand();
      renderActionButtons();
      renderRoomBadges();
      const bannerEl = document.getElementById('live-banner-text');
      if (bannerEl) bannerEl.textContent = roomState.bannerText;
    }

    function bindEvents() {
      const onClick = (id, handler) => {
        const el = document.getElementById(id);
        if (el) {
          el.addEventListener('click', (e) => {
            e.preventDefault();
            safeRun(handler);
          });
        }
      };

      onClick('btn-open-rules', () => openRules());
      onClick('btn-close-rules-x', () => closeRules());
      onClick('btn-close-rules-ok', () => closeRules());
      onClick('btn-leave-to-lobby', () => leaveRoom());

      onClick('btn-create-room', () => createRoomAsHost());
      onClick('btn-join-room', () => joinRoomAsGuest());

      onClick('btn-copy-invite-link', () => copyInviteLink());
      onClick('btn-start-or-next-round', () => hostStartOrNextRound());

      onClick('host-limit-250', () => hostSetLossLimit(250));
      onClick('host-limit-500', () => hostSetLossLimit(500));
      onClick('host-limit-750', () => hostSetLossLimit(750));

      const maxSelect = document.getElementById('room-max-players-select');
      if (maxSelect) {
        maxSelect.addEventListener('change', () => {
          safeRun(() => hostSetMaxPlayers(Number(maxSelect.value)));
        });
      }

      onClick('draw-deck-slot', () => triggerDraw('DECK'));
      onClick('btn-discard', () => triggerDiscard());
      onClick('btn-clear-selection', () => clearSelectedCards());
      onClick('btn-inzili', () => triggerInzili());

      const rulesModal = document.getElementById('rules-modal');
      if (rulesModal) {
        rulesModal.addEventListener('click', (e) => {
          if (e.target === rulesModal) safeRun(() => closeRules());
        });
      }
    }

    window.addEventListener('DOMContentLoaded', () => {
      safeRun(() => {
        bindEvents();
        parseInvitedRoomFromUrl();
        renderLossLimitPill();
      });
    });
